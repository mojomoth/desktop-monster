// The published, weapon-bearing form art remains available for old replays.
// New equipment drawing has separate costume, hands and replaceable prop layers.
import { equipmentJob, equipmentTemplate } from '../../core/equipment.js';
import type { EquipmentItem, WeaponType } from '../../core/equipment.js';
import { equipmentIcon, ITEM_RARITY_COLORS } from './equipment.js';
import { heroFormSprite, HERO_FORM_IDS } from './heroForms.js';
import { drawSprite, registerSprites } from './sprite.js';
import type { DrawSpriteOptions, Sprite, SpriteCanvas } from './sprite.js';

export const HERO_HANDS: readonly (readonly (readonly [number, number])[])[] = [
  [[9, 7], [8, 8]], [[9, 8], [8, 9]], [[5, 6], [5, 7]], [[7, 6], [7, 7]], [[8, 7], [7, 8]],
];
export interface HeroLayers { body: Sprite; hands: Sprite }
const bodies = new Map<string, HeroLayers>();
const blank = (): string[][] => Array.from({ length: 14 }, () => Array<string>(14).fill('.'));

/** Costume panels have explicit authored boundaries, independent of colour.
 * The old sword, staff, book and gun pixels outside them are never copied.
 * Head and torso are rebuilt before hands, so removing a blade cannot erase skin.
 */
function bodyLayers(formId: string): HeroLayers {
  const idle = heroFormSprite(formId), attack = heroFormSprite(formId, true);
  const original = idle.frames[0]!;
  const frames: string[][] = [], hands: string[][] = [];
  for (let pose = 0; pose < 5; pose++) {
    const cells = blank(), hand = blank();
    const source = pose < 2 ? idle.frames[pose]! : attack.frames[pose - 2]!;
    const bob = pose === 1 ? 1 : 0;
    // The face is never sampled from a raised-weapon composite. Keep the full
    // original head trajectory and all its brim/eye/skin colours.
    for (let y = 0; y < 6; y++) {
      const dx = pose === 2 ? (y < 2 ? 1 : 2) : pose === 3 && y >= 2 ? 1 : 0;
      // Polearm costumes have a separate idle tip at x11..13, y3..6.
      // Other families' wide hat/hair edges in those columns are real costume.
      const headWidth = equipmentJob(formId) === 1 && y >= 3 ? 11 : 14;
      for (let x = 0; x < headWidth; x++) if (x + dx < 14) cells[y + bob]![x + dx] = original[y]![x]!;
    }
    // Costume/leg envelopes preserve the authored back, sleeves and footwear.
    // Grip-side prop pixels are excluded even when they share a costume colour.
    for (let y = 6 + bob; y < 14; y++) {
      const limit = y >= 10 + bob ? (pose === 2 || pose === 3 ? 12 : 11)
        : pose === 2 ? 10 : pose === 3 ? (y < 8 ? 6 : 8) : pose === 4 ? 8 : 9;
      for (let x = 0; x <= limit; x++) cells[y]![x] = source[y]![x]!;
    }
    // The abyss knight's diving tank was previously distinguished mostly by
    // its fixed sword. Give the weapon-free costume its own connected tank.
    if (formId === 'h58') for (let y = 7; y <= 11; y++) {
      const dx = pose === 2 ? 1 : 0;
      for (let x = 0; x < 3; x++) cells[y]![x + dx] = y === 7 || y === 11 || x === 0 ? 'e' : 't';
    }
    for (const [x, y] of HERO_HANDS[pose]!) { hand[y]![x] = 's'; cells[y]![x] = '.'; }
    frames.push(cells.map(row => row.join(''))); hands.push(hand.map(row => row.join('')));
  }
  const palette = { ...idle.palette };
  return { body: { w: 14, h: 14, palette, frames }, hands: { w: 14, h: 14, palette, frames: hands } };
}
for (const id of ['h00', ...HERO_FORM_IDS]) bodies.set(id, bodyLayers(id));
registerSprites(Object.fromEntries([...bodies].flatMap(([id, layers]) => [
  [`hero.${id}.body`, layers.body], [`hero.${id}.hands`, layers.hands],
])));
export const heroBodyLayers = (formId: string): HeroLayers => bodies.get(formId) ?? bodies.get('h00')!;

export interface EquippedHeroOptions extends DrawSpriteOptions {
  attacking?: boolean; frame?: number; timeMs?: number;
}
export const EQUIPPED_HERO_PADDING = { x: 10, y: 2 } as const;
interface Pixel { x: number; y: number; color: string }
interface WeaponGeometry { pixels: Pixel[]; bounds: { minX: number; minY: number; maxX: number; maxY: number } }
const pixelCache = new WeakMap<Sprite, Map<number, Pixel[]>>();
const heldCache = new Map<string, WeaponGeometry>();
export function heroEquipmentPose(type: WeaponType | undefined, attacking: boolean, frame: number): number {
  if (!attacking) return frame % 2;
  const step = Math.max(0, Math.min(2, frame));
  if (type === 'gun' || type === 'spear' || type === 'dagger') return [0, 3, 4][step]!;
  if (type === 'staff') return [0, 2, 4][step]!;
  return step + 2;
}
function pixels(sprite: Sprite, frame: number): Pixel[] {
  const frames = pixelCache.get(sprite) ?? new Map<number, Pixel[]>();
  const cached = frames.get(frame); if (cached) return cached;
  const result = (sprite.frames[frame] ?? []).flatMap((row, y) => [...row].flatMap((key, x) =>
    key !== '.' && sprite.palette[key] ? [{ x, y, color: sprite.palette[key]! }] : []));
  frames.set(frame, result); pixelCache.set(sprite, frames); return result;
}

/** Inverse nearest-neighbor rasterization fills every destination grid cell.
 * Forward stamping rounded source points leaves checkerboard holes on diagonals.
 * The complete source pixel areas are fitted above ground BEFORE rasterization;
 * no output pixels are silently discarded to meet the body-height budget.
 */
export function heldWeaponGeometry(item: EquipmentItem, attacking: boolean, frame: number): WeaponGeometry {
  const template = equipmentTemplate(item.templateId), icon = equipmentIcon(item.templateId);
  const pose = heroEquipmentPose(template?.weaponType, attacking, frame);
  const key = `${item.templateId}/${pose}/${attacking ? frame : 'idle'}`;
  const cached = heldCache.get(key); if (cached) return cached;
  if (!template?.weaponType || !icon) return { pixels: [], bounds: { minX: 0, minY: 0, maxX: 0, maxY: 0 } };
  const [hx, hy] = HERO_HANDS[pose]![0]!;
  const type = template.weaponType;
  // Affine columns transform side=(sourceX-7), reach=(12-sourceY).
  let a = .5, b = .5, c = -.5, d = .5, tx = hx, ty = hy;
  if (type === 'gauntlet') {
    a = .75; b = 0; c = 0; d = -.5; tx += attacking && frame === 1 ? 1 : 0; ty += 2;
  } else if (attacking && frame === 1 && type !== 'staff') {
    a = 0; b = 1; c = 1; d = 0;
  } else if (pose === 2) {
    a = .5; b = -.5; c = -.5; d = -.5;
  } else if (type === 'staff' || type === 'spear') {
    // The shaft pivots on the actual hand. Lean only its upper reach outward
    // so the orb/crook clears the face instead of hiding behind the head.
    a = 1; b = .75; c = 0; d = -1;
  }
  const source = pixels(icon, 0);
  const transformedBounds = (): WeaponGeometry['bounds'] => {
    let minX = Infinity, minY = Infinity, maxX = -Infinity, maxY = -Infinity;
    for (const pixel of source) for (const ox of [-.5, .5]) for (const oy of [-.5, .5]) {
      const side = pixel.x - 7 + ox, reach = 12 - pixel.y + oy;
      const x = tx + a * side + b * reach, y = ty + c * side + d * reach;
      minX = Math.min(minX, x); minY = Math.min(minY, y); maxX = Math.max(maxX, x); maxY = Math.max(maxY, y);
    }
    return { minX, minY, maxX, maxY };
  };
  const raw = transformedBounds();
  const fit = Math.min(1, raw.minY < hy ? (hy + .49) / (hy - raw.minY) : 1,
    raw.maxY > hy ? (13.49 - hy) / (raw.maxY - hy) : 1);
  c *= fit; d *= fit; ty = hy + (ty - hy) * fit;
  const bounds = transformedBounds(), result: Pixel[] = [], determinant = a * d - b * c;
  for (let y = Math.floor(bounds.minY); y <= Math.ceil(bounds.maxY); y++) {
    for (let x = Math.floor(bounds.minX); x <= Math.ceil(bounds.maxX); x++) {
      const dx = x - tx, dy = y - ty;
      const sx = Math.floor(7 + (d * dx - b * dy) / determinant + .5);
      const sy = Math.floor(12 - (-c * dx + a * dy) / determinant + .5);
      const color = icon.palette[icon.frames[0]?.[sy]?.[sx] ?? '.'];
      if (color) result.push({ x, y, color });
    }
  }
  const geometry = { pixels: result, bounds }; heldCache.set(key, geometry); return geometry;
}

function layersFor(formId: string, weapon: EquipmentItem | null, opts: EquippedHeroOptions): { body: Pixel[]; hand: Pixel[]; prop: Pixel[] } {
  const frame = opts.frame ?? 0, attacking = opts.attacking ?? false;
  const type = weapon ? equipmentTemplate(weapon.templateId)?.weaponType : undefined;
  const pose = heroEquipmentPose(type, attacking, frame), layers = heroBodyLayers(formId);
  const body = pixels(layers.body, pose), hand = [...pixels(layers.hands, pose)];
  if (!weapon && attacking && frame === 1) {
    // Closed bare fist, not a handle-shaped hand holding an invisible sword.
    const color = layers.hands.palette.s!;
    hand.push({ x: 8, y: 6, color }, { x: 8, y: 7, color }, { x: 9, y: 6, color });
  }
  return { body, hand, prop: weapon ? heldWeaponGeometry(weapon, attacking, frame).pixels : [] };
}

/** Body origin remains 14×14; the weapon mirrors around THAT origin as well. */
export function drawEquippedHero(
  ctx: SpriteCanvas, formId: string, weapon: EquipmentItem | null, x: number, y: number, opts: EquippedHeroOptions = {},
): void {
  const { body, hand, prop } = layersFor(formId, weapon, opts);
  const scale = opts.scale ?? 1;
  const paint = (pixel: Pixel): void => {
    ctx.fillStyle = opts.tint ?? pixel.color;
    ctx.fillRect(x + (opts.flipX ? 13 - pixel.x : pixel.x) * scale, y + pixel.y * scale, scale, scale);
  };
  const rarity = weapon ? equipmentTemplate(weapon.templateId)?.rarity : undefined;
  const occupied = new Set(prop.map(p => `${p.x},${p.y}`));
  if (!opts.tint && (rarity === 'rare' || rarity === 'epic')) {
    const edge = new Map<string, Pixel>();
    for (const p of prop) for (const [dx, dy] of [[-1, 0], [1, 0], [0, -1], [0, 1]]) {
      const px = p.x + dx!, py = p.y + dy!, key = `${px},${py}`;
      if (!occupied.has(key)) edge.set(key, { x: px, y: py,
        color: rarity === 'epic' && (px + py + Math.floor((opts.timeMs ?? 0) / 120)) % 5 === 0 ? '#fff0cc' : ITEM_RARITY_COLORS[rarity] });
    }
    for (const p of edge.values()) paint(p);
  }
  // Put the entire prop behind the body first. Only its unobscured front part
  // and grip reappear on top, so a large headpiece never disappears under a blade.
  for (const p of prop) paint(p);
  for (const p of body) paint(p);
  const face = new Set(body.filter(p => p.y <= 5).map(p => `${p.x},${p.y}`));
  for (const p of prop) if (!face.has(`${p.x},${p.y}`)) paint(p);
  for (const p of hand) paint(p);
}

/** Exact composited pixels, including the equipment contour, for fever and KO. */
export function equippedHeroSprite(formId: string, weapon: EquipmentItem | null, opts: EquippedHeroOptions = {}): Sprite {
  const points = new Map<string, Pixel>();
  const ctx: SpriteCanvas = { fillStyle: '', fillRect(x, y) { points.set(`${x},${y}`, { x, y, color: String(ctx.fillStyle) }); } };
  drawEquippedHero(ctx, formId, weapon, EQUIPPED_HERO_PADDING.x, EQUIPPED_HERO_PADDING.y, { ...opts, scale: 1 });
  const palette: Record<string, string> = {}, keys = new Map<string, string>();
  const rows = Array.from({ length: 18 }, () => Array<string>(36).fill('.'));
  for (const p of points.values()) {
    if (p.x < 0 || p.x >= 36 || p.y < 0 || p.y >= 18) continue;
    if (!keys.has(p.color)) { const key = String.fromCharCode(65 + keys.size); keys.set(p.color, key); palette[key] = p.color; }
    rows[p.y]![p.x] = keys.get(p.color)!;
  }
  return { w: 36, h: 18, palette, frames: [rows.map(row => row.join(''))] };
}

/** Draw the weapon-free costume and hands without allocating a composite sprite. */
export function drawBareHero(ctx: SpriteCanvas, formId: string, pose: number, x: number, y: number, opts?: DrawSpriteOptions): void {
  const layers = heroBodyLayers(formId);
  drawSprite(ctx, layers.body, pose, x, y, opts); drawSprite(ctx, layers.hands, pose, x, y, opts);
}
