function tree(root,prefix=''){
  const found={};for(const name of fs.readdirSync(path.join(root,prefix)).sort()){
    const rel=path.join(prefix,name),p=path.join(root,rel),s=fs.lstatSync(p);
    if(s.isSymbolicLink())found[rel]={link:fs.readlinkSync(p)};
    else if(s.isDirectory())Object.assign(found,tree(root,rel));
    else found[rel]={bytes:s.size,sha256:hash(p)};
  }return found;
}
