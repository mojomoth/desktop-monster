import assert from 'node:assert/strict';
import {readFileSync} from 'node:fs';
import {sourceDigest,evaluationDigest,sha256,ROOT} from '../../../.harness/v5/loop/evidence.mjs';
import {validateArtifacts} from '../../../.harness/v5/loop/audit.mjs';
const read=name=>JSON.parse(readFileSync(new URL(name,import.meta.url)));
const measure=read('./measure.json'),experiment=read('./experiments.json'),review=read('./balance-final-review.json');
const state=read('./final-matrix-03/matrix-state.json');
validateArtifacts(JSON.parse(readFileSync(state.runs[0].path)),measure,sourceDigest());
assert.equal(experiment.adoption.decision,'excluded');
assert.equal(experiment.adoption.numericCriteriaPass,false);
assert.equal(review.sourceDigest,sourceDigest());assert.equal(review.evaluationDigest,evaluationDigest());
assert.equal(review.sha256,sha256(readFileSync(new URL('./experiments.json',import.meta.url))));
assert.equal(review.canonical.sha256,sha256(readFileSync(new URL('./measure.json',import.meta.url))));
assert.equal(review.checks.captureRows,10800);assert.equal(review.canonical.checks.rawSamples,1800);
const protocol=JSON.parse(readFileSync(ROOT+'/docs/v0.6/EVALUATION_PROTOCOL.json'));
for(const record of Object.values(protocol.captureExperiment.fixtureRecords))assert.equal(sha256(readFileSync(record.path)),record.sha256);
const document=readFileSync(ROOT+'/docs/v0.6/EXPERIMENT_RESULTS.md','utf8');
assert(document.includes(sourceDigest())&&document.includes(evaluationDigest())&&document.includes('조건부 제외'));
console.log(JSON.stringify({status:'passed',sourceDigest:sourceDigest(),evaluationDigest:evaluationDigest(),captureRows:10800,canonicalRows:1800,distributions:review.canonical.checks.distributionsRecomputed,decision:experiment.adoption.decision,independentReview:review.kind}));
