import { writeFileSync } from 'node:fs';
import { describe, it } from 'vitest';
import { createEngine, heroReady, mulberry32 } from '../src/core/index.js';
import { heroForm } from '../src/core/hero.js';

const BASE = (r: number) => 50 + 25 * Math.min(100, r);
const PRICE = (r: number, k: number) => BASE(r) * Math.pow(2, Math.min(k, 5));

type Policy = 'baseline' | 'bestEquip';

interface Row { seed: number; policy: Policy; kills: number; coins: number; collected: number;
  reinc: number; equippedBuff: number; bestBuff: number;
  // offline economy of the proposed 각인 재추첨 (side RNG, gold accounted offline)
  spend: number; rerolls: number; blocked: number; chasedBest: number; chasedEquipped: number;
  freeBestParty: number; chaseBestParty: number; satOffer: number; spendUntilSat: number; restIfRational: number }

function run(seed: number, policy: Policy): Row {
  const engine = createEngine(null, mulberry32(seed));
  const side = mulberry32(seed * 7919 + 13); // independent stream for the simulated buff redraw
  let spend = 0, rerolls = 0, blocked = 0;
  const chased: Record<string, number> = {}; // formId -> best engraving under the chase policy
  let chasedEquipped = 0;
  let spendUntilSat = 0; let satOffer = 0; let offerNo = 0;
  const freeParty: number[] = []; const chaseParty: number[] = [];
  for (let sec = 1; sec <= 1800; sec++) {
    for (let i = 0; i < 2; i++) engine.attack('keyboard');
    engine.tick(1000);
    const st = engine.getState();
    if (heroReady(st.level, st.hero)) {
      engine.apply({ type: 'heroOffer' });
      const hero = engine.getState().hero!;
      const r = hero.reincarnations;
      // ---- offline: 각인 재추첨 "25를 노린다" policy on this offer ----
      offerNo++;
      let coins = engine.getState().coins - spend;
      const cards = hero.choices.map((c) => c.buffPercent);
      let k = 0;
      for (;;) {
        if (cards[0]! >= 25) break;
        const p = PRICE(r, k);
        if (coins < p) { blocked++; break; }
        coins -= p; spend += p; rerolls++; k++;
        for (let i = 0; i < 3; i++) cards[i] = Math.max(cards[i]!, 10 + Math.min(15, Math.floor(side.next() * 16)));
      }
      const pick = hero.choices[0]!;
      chased[pick.formId] = Math.max(chased[pick.formId] ?? 0, cards[0]!);
      chasedEquipped = chased[pick.formId]!;
      if (heroForm(pick.formId)!.buff === 'party') { chaseParty.push(cards[0]!); freeParty.push(pick.buffPercent); }
      // saturation: once ANY party form in the chase collection is 25, further gold buys no power
      if (satOffer === 0 && Math.max(0, ...chaseParty) >= 25) { satOffer = offerNo; spendUntilSat = spend; }
      // ---- real engine choice (unchanged, first card) ----
      engine.apply({ type: 'heroChoose', formId: pick.formId, offerSerial: hero.offerSerial });
      if (policy === 'bestEquip') {
        const h = engine.getState().hero!;
        const best = [...h.collection].sort((a, b) =>
          (heroForm(b.formId)!.buff === 'party' ? b.buffPercent : 0) - (heroForm(a.formId)!.buff === 'party' ? a.buffPercent : 0))[0];
        if (best && best.formId !== h.equipped.formId) engine.apply({ type: 'heroEquip', formId: best.formId });
      }
    }
  }
  const st = engine.getState();
  const h = st.hero!;
  return { seed, policy, kills: st.killCount, coins: st.coins, collected: h.collection.length,
    reinc: h.reincarnations, equippedBuff: h.equipped.buffPercent,
    bestBuff: Math.max(0, ...h.collection.map((c) => c.buffPercent)),
    spend, rerolls, blocked, chasedBest: Math.max(0, ...Object.values(chased)), chasedEquipped,
    freeBestParty: Math.max(0, ...freeParty), chaseBestParty: Math.max(0, ...chaseParty),
    satOffer, spendUntilSat, restIfRational: st.coins - spendUntilSat };
}

const q = (v: number[]) => { const s = [...v].sort((a, b) => a - b);
  return { p10: s[Math.floor((s.length - 1) * .1)]!, p50: s[Math.floor((s.length - 1) * .5)]!, p90: s[Math.floor((s.length - 1) * .9)]! }; };

describe('critic r3 scratch', () => {
  it('measures the value of a higher engraving and the offline economy', () => {
    const rows: Row[] = [];
    for (const policy of ['baseline', 'bestEquip'] as const)
      for (let seed = 1; seed <= 100; seed++) rows.push(run(seed, policy));
    const base = rows.filter((r) => r.policy === 'baseline');
    const best = rows.filter((r) => r.policy === 'bestEquip');
    const wins = base.filter((b, i) => best[i]!.kills > b.kills).length;
    const ties = base.filter((b, i) => best[i]!.kills === b.kills).length;
    const out = {
      baseline: { kills: q(base.map((r) => r.kills)), coins: q(base.map((r) => r.coins)),
        equippedBuff: q(base.map((r) => r.equippedBuff)), bestBuff: q(base.map((r) => r.bestBuff)),
        collected: q(base.map((r) => r.collected)) },
      bestEquip: { kills: q(best.map((r) => r.kills)), equippedBuff: q(best.map((r) => r.equippedBuff)),
        collected: q(best.map((r) => r.collected)) },
      pairedKills: { bestEquipWins: wins, ties, losses: 100 - wins - ties,
        medianDelta: q(base.map((b, i) => best[i]!.kills - b.kills)) },
      chase: { spend: q(base.map((r) => r.spend)), rerolls: q(base.map((r) => r.rerolls)),
        remaining: q(base.map((r) => r.coins - r.spend)), blockedSeeds: base.filter((r) => r.blocked > 0).length,
        chasedBest: q(base.map((r) => r.chasedBest)), chasedEquipped: q(base.map((r) => r.chasedEquipped)) },
      party: { freeBestParty: q(base.map((r) => r.freeBestParty)), chaseBestParty: q(base.map((r) => r.chaseBestParty)),
        freeReaches25: base.filter((r) => r.freeBestParty >= 25).length,
        chaseReaches25: base.filter((r) => r.chaseBestParty >= 25).length,
        chaseStrictlyBetter: base.filter((r) => r.chaseBestParty > r.freeBestParty).length,
        gain: q(base.map((r) => r.chaseBestParty - r.freeBestParty)),
        satOffer: q(base.filter((r) => r.satOffer > 0).map((r) => r.satOffer)),
        satSeeds: base.filter((r) => r.satOffer > 0).length,
        spendUntilSat: q(base.filter((r) => r.satOffer > 0).map((r) => r.spendUntilSat)),
        rationalRemaining: q(base.filter((r) => r.satOffer > 0).map((r) => r.restIfRational)) },
    };
    writeFileSync('/tmp/critic_r3.json', JSON.stringify(out, null, 2));
    process.stdout.write(JSON.stringify(out, null, 2) + '\n');
  }, 900000);
});
